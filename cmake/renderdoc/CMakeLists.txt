cmake_minimum_required(VERSION 3.22)
project(shadps4_renderdoc_turnip LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "The RenderDoc Turnip loader requires Android arm64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../externals/libadrenotools adrenotools)
add_library(renderdoc_turnip SHARED android_turnip_loader.cpp)
target_link_libraries(renderdoc_turnip PRIVATE adrenotools dl log)
target_compile_options(renderdoc_turnip PRIVATE -fvisibility=hidden)
target_link_options(renderdoc_turnip PRIVATE -Wl,--no-undefined -Wl,--exclude-libs,ALL)
