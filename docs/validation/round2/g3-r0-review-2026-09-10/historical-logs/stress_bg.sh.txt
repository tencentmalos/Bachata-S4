for i in $(seq 25 70); do
  adb -s 9c2841a4 shell "timeout 120 /data/local/tmp/g3_repro/guest; echo __EXIT__=\$?" > /tmp/g3-r0-repro/stress$i.txt 2>&1
  fail=$(grep -cE '\[g24\] FAIL' /tmp/g3-r0-repro/stress$i.txt)
  echo "run$i: exit=$(grep -o '__EXIT__=[0-9]*' /tmp/g3-r0-repro/stress$i.txt|tail -1) stagefails=$fail"
  if [ "$fail" != "0" ]; then
    echo "=== REPRODUCED run$i ==="
    grep -E '\[g24\] (FAIL|detail)' /tmp/g3-r0-repro/stress$i.txt
    break
  fi
done
echo "STRESS_DONE"
