cmake_minimum_required(VERSION 3.22)
project(g3_n4_review LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(REVIEW_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(V0_ENABLE_FEX ON CACHE BOOL "" FORCE)
set(V0_BUILD_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory("${REVIEW_ROOT}/cmake/fex" backend)
foreach(name hle_probe gate_timeout_probe)
    add_executable(${name} "${name}.cpp")
    target_include_directories(${name} PRIVATE "${REVIEW_ROOT}/tests/guest_cpu" "${V0_FIXTURE_DIR}")
    target_compile_definitions(${name} PRIVATE GUEST_CPU_TEST_HOOKS=1)
    target_link_libraries(${name} PRIVATE guest_cpu_fex)
    target_link_options(${name} PRIVATE -Wl,--build-id=sha1)
    add_dependencies(${name} guest_execution_tests)
endforeach()
