cmake_minimum_required(VERSION 3.16)
project(orbis_sync_performance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT ../../.. ABSOLUTE BASE_DIR ${CMAKE_CURRENT_SOURCE_DIR})
find_package(Threads REQUIRED)
add_subdirectory(${ROOT}/src/core/guest_cpu ${CMAKE_CURRENT_BINARY_DIR}/guest_cpu)
add_executable(guest_sync_performance ../guest_sync_performance.cpp
    ${ROOT}/src/common/native_clock.cpp ${ROOT}/src/common/rdtsc.cpp
    ${ROOT}/src/core/libraries/kernel/sync/mutex.cpp)
target_include_directories(guest_sync_performance PRIVATE ${ROOT}/externals/spdlog/include ${ROOT}/externals/fmt/include)
target_link_libraries(guest_sync_performance PRIVATE guest_cpu_api Threads::Threads)
if(ANDROID)
    target_link_libraries(guest_sync_performance PRIVATE log)
endif()
add_executable(guest_sync_isolation_tests ../guest_sync_isolation_tests.cpp)
target_link_libraries(guest_sync_isolation_tests PRIVATE guest_cpu_api Threads::Threads)
enable_testing()
add_test(NAME guest_sync_isolation COMMAND guest_sync_isolation_tests)
add_test(NAME guest_sync_unrelated_wake COMMAND guest_sync_performance --assert-isolation)
