cmake_minimum_required(VERSION 3.22)
project(shad_pinyin_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(ROOT ../.. ABSOLUTE)
add_subdirectory(${ROOT}/cmake/pinyin pinyin)
add_executable(pinyin_tests pinyin_tests.cpp ${ROOT}/src/core/libraries/ime/pinyin_composer.cpp)
target_include_directories(pinyin_tests PRIVATE ${ROOT}/src/core/libraries/ime)
target_link_libraries(pinyin_tests PRIVATE shad_pinyin_dictionary)
enable_testing()
get_target_property(DICTIONARY spatial::foundation_pinyin FOUNDATION_PINYIN_DICTIONARY)
add_test(NAME pinyin COMMAND pinyin_tests ${DICTIONARY} ${CMAKE_CURRENT_BINARY_DIR}/user.dat)
