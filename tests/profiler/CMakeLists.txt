cmake_minimum_required(VERSION 3.22)
project(shadps4_profiler_tests LANGUAGES C CXX)
enable_testing()
get_filename_component(repo "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${repo}/foundation/modules/profiler_ring" foundation-ring)
set(COMMON_PROFILER_SOURCE "${repo}/src/common/profiler.cpp" CACHE FILEPATH "Profiler under test")
add_executable(present_cadence_tests present_cadence_tests.cpp "${COMMON_PROFILER_SOURCE}")
target_compile_features(present_cadence_tests PRIVATE cxx_std_20)
target_compile_definitions(present_cadence_tests PRIVATE SHADPS4_PROFILER_RING)
target_include_directories(present_cadence_tests PRIVATE "${repo}/src")
target_link_libraries(present_cadence_tests PRIVATE spatial::foundation_profiler_ring)
add_test(NAME present_cadence_tests COMMAND present_cadence_tests "${CMAKE_CURRENT_BINARY_DIR}/evidence")
